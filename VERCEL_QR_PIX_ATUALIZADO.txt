POST /api/qr

Exemplo:
{
  "amount": 0.01,
  "description": "Google AI Pro",
  "name": "Samuel",
  "phone": "(99) 99999-9999",
  "email": "SAMUEL@HOTMAIL.COM"
}

qr_data, qrData, qrDataUrl, pixCopyPaste, orderId, status, expiresAt.

MP_WEBHOOK_SECRET=SEU_SEGREDO
PRODUCT_NAME=Google AI Pro
PRODUCT_PRICE=0.01
DELIVERY_CODE=GEMINI-DEMO-001
